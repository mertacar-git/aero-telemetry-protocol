cmake_minimum_required(VERSION 3.16)
project(aero_telemetry LANGUAGES CXX)
add_library(aero_telemetry INTERFACE)
target_include_directories(aero_telemetry INTERFACE cpp/include)
target_compile_features(aero_telemetry INTERFACE cxx_std_17)
add_executable(codec_test cpp/tests/test_codec.cpp)
target_link_libraries(codec_test PRIVATE aero_telemetry)
enable_testing()
add_test(NAME codec COMMAND codec_test)
